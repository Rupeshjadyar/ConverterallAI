Phase 3: Vanilla JavaScript Interaction

We wire up client-side form validation and success handling without any third-party dependencies: Contact form submission handle karne ke liye vanilla JavaScript add karein:

JavaScript — app.js 
document.getElementById('portfolio-contact').addEventListener('submit', function(e) {
  e.preventDefault(); // Prevent page reload

  const name = document.getElementById('client-name').value.trim();
  const email = document.getElementById('client-email').value.trim();
  const msg = document.getElementById('client-msg').value.trim();
  const feedback = document.getElementById('form-feedback');

  if (!name || !email || !msg) {
    feedback.style.display = 'block';
    feedback.style.color = '#ef4444';
    feedback.textContent = 'Please fill out all required fields!';
    return;
  }

  // Simulated submission
  feedback.style.display = 'block';
  feedback.style.color = '#10b981';
  feedback.textContent = `Thank you, ${name}! Your message has been received.`;
  this.reset();
});
